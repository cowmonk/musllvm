cmake_minimum_required(VERSION 3.20)
project(musllvm_smoke LANGUAGES C CXX)
add_executable(hello-c ../hello.c)
set_target_properties(hello-c PROPERTIES C_STANDARD 99)
target_compile_options(hello-c PRIVATE -pthread)
target_link_options(hello-c PRIVATE -pthread)
add_executable(hello-cxx ../hello.cpp)
set_target_properties(hello-cxx PROPERTIES CXX_STANDARD 20)
target_compile_options(hello-cxx PRIVATE -pthread)
target_link_options(hello-cxx PRIVATE -pthread)
