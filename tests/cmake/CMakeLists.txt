cmake_minimum_required(VERSION 3.20)
project(musllvm_smoke LANGUAGES C CXX)
add_executable(hello-c ../hello.c)
set_target_properties(hello-c PROPERTIES C_STANDARD 99)
add_executable(hello-cxx ../hello.cpp)
add_executable(hello-cxx-static ../hello.cpp)
set_target_properties(hello-cxx hello-cxx-static PROPERTIES CXX_STANDARD 20)
target_link_options(hello-cxx-static PRIVATE -static)
foreach(target hello-c hello-cxx hello-cxx-static)
  target_compile_options(${target} PRIVATE -pthread)
  target_link_options(${target} PRIVATE -pthread)
endforeach()
